-- Territories behind target, ranked by revenue at risk
WITH wk AS (
  SELECT territory,
         SUM(net_sales)    AS actual,
         SUM(target_sales) AS target
  FROM   fact_sales
  WHERE  week_id = @this_week
  GROUP BY territory
)
SELECT territory,
       actual / target - 1 AS gap_pct,
       target - actual     AS at_risk
FROM   wk
WHERE  actual < target
ORDER BY at_risk DESC;
